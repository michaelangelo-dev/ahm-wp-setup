<script>
(() => {
    const config = {
        init: function () {
            this.handleScroll()
            this.bindEvents()
            this.devScripts()

            this.observeDateInputs()
            this.formSubmitSpinner()
            this.addFormSpinnerSyle()
            this.headerMenuHoverEvents()
            this.initIntlTelInput()

            // this.oddEvenGridItemGridItems()
            // this.slidePauseOnLightbox()
            // this.imageCarousel()
            // this.tocOverride()
            // this.observeOthersDropdown()
            // this.initVideoTooltip()
        },

        handleScroll: function () {
            const isScrolled = window.scrollY >= 30
            document.body.classList.toggle('is-scrolled', isScrolled)
        },

        bindEvents: function () {
            window.addEventListener('resize', () => this.handleScroll())
            window.addEventListener('scroll', () => this.handleScroll(), { passive: true })
            window.addEventListener('keydown', (e) => this.devScripts(e))
        },

        devScripts: function (e) {
            if (! e) return

            const isModifier = e.ctrlKey || e.metaKey

            // Add red outline to all elements for debugging
            if (isModifier && e.key === 'ArrowDown') {
                e.preventDefault()

                const allElements = document.querySelectorAll('*')
                const isOutlined = document.body.dataset.outlineActive === 'true'

                if (isOutlined) {
                    allElements.forEach(el => {
                        el.style.outline = ''
                    })
                    document.body.dataset.outlineActive = 'false'
                } else {
                    allElements.forEach(el => {
                        el.style.outline = '1px solid red'
                    })
                    document.body.dataset.outlineActive = 'true'
                }
            }

            // Reload page with ?nowprocket to test without Rocket Loader
            if (isModifier && e.key === 'ArrowUp') {
                e.preventDefault()
                const url = new URL(window.location.href)

                if (!url.searchParams.has('nowprocket')) {
                    url.searchParams.set('nowprocket', '')
                    window.location.href = url.toString()
                }
            }
        },

        addFormSpinnerSyle: function() {
            const style = document.createElement('style')
            const css = `
                /* Create the Spinner using a pseudo-element */
                .elementor-form .elementor-button.is-loading::after {
                    content: "";
                    position: absolute;
                    width: 18px;
                    height: 18px;
                    top: 0;
                    left: 0;
                    right: 0;
                    bottom: 0;
                    margin: auto;
                    border: 2px solid rgba(255, 255, 255, 0.3);
                    border-top-color: #fff;
                    border-radius: 50%;
                    animation: spin 0.6s linear infinite;
                }

                .elementor-form .elementor-button.is-loading .elementor-button-content-wrapper {
                    opacity: 0;
                }

                /* Loading Animation Keyframes */
                @keyframes spin {
                    to {
                        transform: rotate(360deg);
                    }
                }
            `

            style.textContent = css
            document.head.appendChild(style)
        },

        headerMenuHoverEvents: function () {
            const menuItems = document.querySelectorAll('.header-menu-item')
            const target = document.querySelector('.header-menu-item-image')

            if (! target || ! menuItems.length) return

            const computedTargetImage = window.getComputedStyle(target).backgroundImage
            const defaultBg = computedTargetImage !== 'none' ? computedTargetImage : `url('/wp-content/uploads/2026/05/ImageWithFallback-4.webp')`

            menuItems.forEach(item => {
                const img = item.querySelector('img')
                if (! img) return

                // Elementor-safe image source resolution
                const imgUrl =
                    img.getAttribute('data-src') ||
                    img.getAttribute('data-lazy-src') ||
                    img.currentSrc ||
                    img.src

                item.addEventListener('mouseenter', () => {
                    if (!imgUrl) return
                    target.style.backgroundImage = `url("${imgUrl}")`
                })

                item.addEventListener('mouseleave', () => {
                    target.style.backgroundImage = defaultBg
                })
            })
        },

        formSubmitSpinner: function() {
            document.addEventListener('submit', function(event) {
                const form = event.target

                if (!form || !form.classList.contains('elementor-form')) return

                const button = form.querySelector('.elementor-button[type="submit"]')
                if (button) {
                    button.classList.add('is-loading')
                    button.disabled = true
                }

                const observer = new MutationObserver(function(mutations, observerInstance) {
                    let shouldStopSpinner = false

                    for (const mutation of mutations) {
                        if (mutation.type === 'childList') {
                            mutation.addedNodes.forEach(node => {
                                if (node.nodeType === 1) {
                                    if (
                                        node.classList.contains('elementor-message') ||
                                        node.querySelector('.elementor-message')
                                    ) {
                                        shouldStopSpinner = true
                                    }
                                }
                            })
                        }
                        
                        if (mutation.type === 'attributes' && mutation.attributeName === 'class') {
                            const target = mutation.target
                            const hasErrorClass = target.classList.contains('elementor-error')
                            const hadErrorClass = mutation.oldValue ? mutation.oldValue.includes('elementor-error') : false
                            
                            if (hasErrorClass && !hadErrorClass) {
                                shouldStopSpinner = true
                            }
                        }
                    }

                    if (shouldStopSpinner) {
                        if (button) {
                            button.classList.remove('is-loading')
                            button.disabled = false
                        }
                        observerInstance.disconnect()
                    }
                })

                observer.observe(form, { 
                    childList: true, 
                    subtree: true, 
                    attributes: true, 
                    attributeFilter: ['class'],
                    attributeOldValue: true
                })
            })
        },

        initIntlTelInput: function() {
            const initTelInputs = () => {
                const telInputs = document.querySelectorAll('input[type="tel"]')
                
                if (! telInputs.length) return
                
                const applyIntlTelInput = () => {
                    telInputs.forEach(input => {
                        if (input.dataset.intlTelInputInitialized) return
                        
                        window.intlTelInput(input, {
                            initialCountry: 'auto',
                            geoIpLookup: callback => {
                                fetch('https://ipapi.co/json')
                                    .then(res => res.json())
                                    .then(data => callback(data.country_code))
                                    .catch(() => callback('us'))
                            },
                            // Ensure utilsScript is loaded for formatting/validation
                            utilsScript: 'https://cdn.jsdelivr.net/npm/intl-tel-input@18.2.1/build/js/utils.js'
                        })
                        
                        input.dataset.intlTelInputInitialized = 'true'
                        
                        const parent = input.closest('.elementor-field-type-tel')
                        if (parent) parent.classList.add('intlTelInput-initiated')
                    })
                }

                if (typeof window.intlTelInput === 'function') {
                    applyIntlTelInput()
                    return
                }

                if (document.getElementById('intl-tel-input-js-script')) return

                const cssLink = document.createElement('link')
                cssLink.rel = 'stylesheet'
                cssLink.href = 'https://cdn.jsdelivr.net/npm/intl-tel-input@18.2.1/build/css/intlTelInput.css'
                document.head.appendChild(cssLink)

                const jsScript = document.createElement('script')
                jsScript.id = 'intl-tel-input-js-script'
                jsScript.src = 'https://cdn.jsdelivr.net/npm/intl-tel-input@18.2.1/build/js/intlTelInput.min.js'
                jsScript.onload = applyIntlTelInput
                document.head.appendChild(jsScript)
            }
            
            // Run immediately
            initTelInputs()
            
            // Re-run for dynamically added elements (like popups)
            const observer = new MutationObserver((mutations) => {
                const hasNewNodes = mutations.some(m => m.addedNodes.length > 0)
                if (hasNewNodes) initTelInputs()
            })
            
            observer.observe(document.body, { childList: true, subtree: true })
        },

        applyDateFieldSettings: function(el) {
            const fields = {
                // Field for today or future dates
                'form-field-prferred_date': { minDate: 'today', maxDate: null },

                // Field for today or past dates
                // 'form-field-field_3a2b774':   { minDate: null, maxDate: 'today' }
            }
            const fieldId = el.getAttribute('id')
            const settings = fields[fieldId]

            const checkInstance = setInterval(() => {
                if (el._flatpickr) {
                    const instance = el._flatpickr

                    // Manually remove the native validation
                    el.removeAttribute('pattern')

                    if (settings) {
                        // For today or future dates
                        if (settings.minDate) instance.set('minDate', settings.minDate)

                        // For today or past dates
                        if (settings.maxDate) instance.set('maxDate', settings.maxDate)

                        instance.set('allowInput', true)
                        instance.set('dateFormat', 'd/m/Y')

                        if (el.value) {
                            instance.setDate(el.value, true, 'd/m/Y')
                        }

                        el.setAttribute('placeholder', 'DD/MM/YYYY')
                        clearInterval(checkInstance)
                    }
                }
            }, 50)

            setTimeout(() => clearInterval(checkInstance), 5000);
        },

        observeDateInputs: function() {
            const self = this
            const checkExist = setInterval(() => {
                const inputs = document.querySelectorAll('.flatpickr-input')

                if (inputs.length > 0) {
                    let allInitialized = true

                    inputs.forEach((el) => {
                        if (el._flatpickr && !el.dataset.formatFixed) {
                            self.applyDateFieldSettings(el)
                            el.dataset.formatFixed = "true"
                        } else if (!el._flatpickr) {
                            allInitialized = false
                        }
                    })

                    if (allInitialized) clearInterval(checkExist)
                }
            }, 100)

            setTimeout(() => clearInterval(checkExist), 10000)
        },

        oddEvenGridItemGridItems: function() {
            const targetContainer = document.querySelector('.with-odd-even-grid .elementor-loop-container')
            if (!targetContainer) return

            const applyGridClasses = () => {
            const gridItems = targetContainer.querySelectorAll(':scope > .e-loop-item')
                gridItems.forEach((item, index) => {
                    if (index % 2 === 0) {
                        item.classList.add('is-odd')
                        item.classList.remove('is-even')
                    } else {
                        item.classList.add('is-even')
                        item.classList.remove('is-odd')
                    }
                })
            }

            applyGridClasses()

            const observer = new MutationObserver(applyGridClasses)
            observer.observe(targetContainer, { childList: true })
        },

        slidePauseOnLightbox: function () {
            const carousels = document.querySelectorAll('.some-selector .e-n-carousel')

            if (! carousels.length) return

            const toggleAutoplay = (action) => {
                carousels.forEach(carousel => {
                    const swiper = carousel.swiper
                    if (swiper && swiper.autoplay) {
                        action === 'stop' ? swiper.autoplay.stop() : swiper.autoplay.start()
                    }
                })
            }

            const isLightbox = (node) => {
                return node.nodeType === 1 && (node.id === 'elementor-lightbox' || node.classList.contains('dialog-lightbox-widget'))
            }

            const observer = new MutationObserver(mutations => {
                mutations.forEach(mutation => {
                    mutation.addedNodes.forEach(node => {
                        if (isLightbox(node)) toggleAutoplay('stop')
                    })

                    mutation.removedNodes.forEach(node => {
                        if (isLightbox(node)) toggleAutoplay('start')
                    })

                    if (mutation.type === 'attributes' && isLightbox(mutation.target)) {
                        const target = mutation.target
                        const isHidden = window.getComputedStyle(target).display === 'none' || target.getAttribute('aria-hidden') === 'true'
                        isHidden ? toggleAutoplay('start') : toggleAutoplay('stop')
                    }
                })
            })

            observer.observe(document.body, {
                childList: true,
                subtree: true,
                attributes: true,
                attributeFilter: ['style', 'class', 'aria-hidden']
            })
        },

        imageCarousel: function () {
            function applyCarousel() {
                document.querySelectorAll('.image-carousel .swiper').forEach(function (el) {
                    const swiper = el.swiper
                    if (!swiper) return

                    swiper.params.slidesPerView = 4
                    swiper.params.spaceBetween = 28
                    swiper.params.loop = true

                    swiper.params.breakpoints = {
                        768: {
                            slidesPerView: 5
                        },
                        1025: {
                            slidesPerView: 7,
                            spaceBetween: 45
                        },
                        1440: {
                            slidesPerView: 9,
                            spaceBetween: 75
                        }
                    }

                    swiper.update()
                })
            }

            function initCarousel() {
                const check = setInterval(function () {
                    const swipers = document.querySelectorAll('.image-carousel .swiper')
                    const ready = Array.from(swipers).some(el => el.swiper)
                    if (!ready) return

                    clearInterval(check)
                    applyCarousel()
                }, 200)
            }

            initCarousel()
        },

        tocOverride: function () {
            document.addEventListener('click', function(e) {
                const link = e.target.closest('.elementor-toc__list-item-text')

                if (!link) return

                const targetHref = link.getAttribute('href')

                if (!targetHref || !targetHref.startsWith('#')) return

                const targetHeading = document.querySelector(targetHref)

                if (targetHeading) {
                    e.preventDefault()

                    const stickyHeaderHeight = 148
                    const headingPosition = targetHeading.getBoundingClientRect().top + window.scrollY
                    const finalPosition = headingPosition - stickyHeaderHeight

                    window.scrollTo({
                        top: finalPosition,
                        behavior: 'smooth'
                    })
                }
            })
        },

        observeOthersDropdown: function () {
            // Map the select element selector to the dependent "Other" field selector
            const fieldsMap = {
                // '#form-field-field_b9ddd79': '.elementor-field-group-field_786199a'
            }

            const toggleOtherField = (selectElement, otherElement) => {
                const otherWrapper = otherElement.closest('.elementor-field-group') || otherElement
                const otherWrapperField = otherWrapper.querySelector('input[type="text"]')
                const isOthersSelected = selectElement.value === 'Others' || selectElement.value === 'others'

                if (isOthersSelected) {
                    otherWrapper.style.display = 'block'
                    otherWrapperField.setAttribute('required', 'required')
                    otherWrapperField.setAttribute('aria-required', 'true')
                } else {
                    otherWrapper.style.display = 'none'
                    otherWrapperField.removeAttribute('required')
                    otherWrapperField.removeAttribute('aria-required')
                }
            }

            const setupFields = () => {
                Object.entries(fieldsMap).forEach(([selectSelector, otherSelector]) => {
                    const selectElement = document.querySelector(selectSelector)
                    const otherElement = document.querySelector(otherSelector)

                    if (selectElement && otherElement) {
                        toggleOtherField(selectElement, otherElement)

                        if (!selectElement.dataset.othersBound) {
                            selectElement.addEventListener('change', () => {
                                toggleOtherField(selectElement, otherElement)
                            })
                            selectElement.dataset.othersBound = 'true'
                        }
                    }
                })
            }

            const targetNode = document.body
            const observerConfig = { childList: true, subtree: true }

            const callback = (mutationsList) => {
                let shouldCheck = false
                for (const mutation of mutationsList) {
                    if (mutation.addedNodes.length > 0) {
                        shouldCheck = true
                        break
                    }
                }
                if (shouldCheck) {
                    setupFields()
                }
            }

            const observer = new MutationObserver(callback)
            observer.observe(targetNode, observerConfig)

            setupFields()
        },
        initVideoTooltip: function () {
            const createTooltipElement = (triggerEl) => {
                // If class is placed directly on an <img> tag, attach tooltip to parent container
                const container = triggerEl.tagName.toLowerCase() === 'img' 
                    ? triggerEl.parentElement 
                    : triggerEl

                const computedStyle = window.getComputedStyle(container)
                if (computedStyle.position === 'static') {
                    container.style.position = 'relative'
                }

                let tooltip = container.querySelector('.video-unavailable-tooltip')
                if (!tooltip) {
                    const customMsg = triggerEl.getAttribute('data-tooltip-msg') || container.getAttribute('data-tooltip-msg') || "Video isn't available yet"
                    
                    tooltip = document.createElement('div')
                    tooltip.className = 'video-unavailable-tooltip'
                    tooltip.setAttribute('role', 'status')
                    tooltip.setAttribute('aria-live', 'polite')
                    tooltip.innerHTML = `
                        <svg xmlns="http://www.w3.org/2000/svg" width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                            <circle cx="12" cy="12" r="10"></circle>
                            <line x1="12" y1="8" x2="12" y2="12"></line>
                            <line x1="12" y1="16" x2="12.01" y2="16"></line>
                        </svg>
                        <span class="tooltip-text">${customMsg}</span>
                    `
                    container.appendChild(tooltip)
                }

                return container
            }

            /**
             * Pre-renders tooltips for all trigger elements on the page
             */
            const setupAllTooltips = () => {
                const triggers = document.querySelectorAll('.video-tooltip-trigger')
                triggers.forEach(el => createTooltipElement(el))
            }

            // 1. Pre-render immediately on load
            setupAllTooltips()

            // 2. Watch for dynamically added triggers (e.g. popups, lazy loaded sections)
            const observer = new MutationObserver((mutations) => {
                const hasAddedNodes = mutations.some(m => m.addedNodes.length > 0)
                if (hasAddedNodes) setupAllTooltips()
            })
            observer.observe(document.body, { childList: true, subtree: true })

            /**
             * Persistent click handler with 3-second auto-dismiss
             * @param {HTMLElement} triggerEl 
             */
            const showTooltip = (triggerEl) => {
                const container = createTooltipElement(triggerEl)
                const tooltip = container.querySelector('.video-unavailable-tooltip')

                if (!tooltip) return

                if (activeTimeouts.has(container)) {
                    clearTimeout(activeTimeouts.get(container))
                    tooltip.classList.remove('is-active')
                    void tooltip.offsetWidth
                }

                tooltip.classList.add('is-active')

                const timeoutId = setTimeout(() => {
                    tooltip.classList.remove('is-active')
                    activeTimeouts.delete(container)
                }, 3000)

                activeTimeouts.set(container, timeoutId)
            }

            // 3. Delegated click listener
            document.body.addEventListener('click', (e) => {
                const trigger = e.target.closest('.video-tooltip-trigger')
                if (trigger) {
                    e.preventDefault()
                    showTooltip(trigger)
                }
            })
        },
    }

    document.addEventListener('DOMContentLoaded', () => {
        config.init()
    })
})()
</script>
